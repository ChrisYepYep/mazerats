# Maze Rats — https://mazerats.net
#
# The archive itself is meant to be found: someone half-remembering a maze
# they walked two years ago should be able to search for it and land here.
# What follows is everything that isn't the archive.

User-agent: *
Allow: /

# The admin panel (already noindex in its own markup, said again here so a
# crawler never has to fetch it to find that out).
Disallow: /admin
Disallow: /admin.html

# An internal reference sheet for the Volter Goldfish font's picture
# glyphs. Useful when building the site, meaningless in a search result.
Disallow: /glyphs.html

# The Habbo Hogwarts map. Reachable by anyone with the link, but still
# being drawn — half its rooms have no picture and no description yet, and
# a search result pointing at that is worse than no search result at all.
# Lift this (and the noindex in wizard.html's own markup) when it is ready
# to be found.
Disallow: /wizard
Disallow: /wizard.html

# /guess was disallowed while the daily game was an unfinished experiment.
# It is a real feature now — scored, with leaderboards and accounts behind
# it — and a real address people paste at each other, so it is worth being
# findable. It still serves the homepage with the game open, so a crawler
# reaches nothing here it could not already reach at /home.html; and the
# page a searcher lands on is the game itself, which is exactly what the
# link promised. The day's answers are not in the HTML at all — they are
# derived in the browser from the date — so there is nothing datable for an
# index to go stale on.

# Project tooling and the function endpoints. Both already 404 at the edge
# (see netlify.toml); this saves the crawl.
Disallow: /tools/
Disallow: /.netlify/

Sitemap: https://mazerats.net/sitemap.xml
